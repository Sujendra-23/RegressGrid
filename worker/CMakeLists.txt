cmake_minimum_required(VERSION 3.22)
project(regressgrid_worker VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(RG_SANITIZERS "Enable address and undefined-behavior sanitizers" OFF)
option(RG_TSAN "Enable ThreadSanitizer (exclusive with RG_SANITIZERS)" OFF)
option(RG_BENCH "Build micro-benchmarks" ON)
# Cross builds cannot run the target-architecture protoc/grpc plugin; point at host tools.
set(RG_HOST_PROTOC "" CACHE FILEPATH "Host protoc for cross builds")
set(RG_HOST_GRPC_PLUGIN "" CACHE FILEPATH "Host grpc_cpp_plugin for cross builds")
option(RG_TIDY "Enable clang-tidy" OFF)
find_package(Threads REQUIRED)
find_package(gRPC CONFIG REQUIRED)
find_package(Protobuf REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
set(PROTO_ROOT ${CMAKE_CURRENT_SOURCE_DIR}/../proto)
set(GENERATED ${CMAKE_CURRENT_BINARY_DIR}/generated)
file(MAKE_DIRECTORY ${GENERATED})
if(RG_HOST_PROTOC AND RG_HOST_GRPC_PLUGIN)
 set(RG_PROTOC_CMD ${RG_HOST_PROTOC})
 set(RG_GRPC_PLUGIN ${RG_HOST_GRPC_PLUGIN})
else()
 set(RG_PROTOC_CMD protobuf::protoc)
 set(RG_GRPC_PLUGIN $<TARGET_FILE:gRPC::grpc_cpp_plugin>)
endif()
add_custom_command(OUTPUT ${GENERATED}/grid.pb.cc ${GENERATED}/grid.pb.h ${GENERATED}/grid.grpc.pb.cc ${GENERATED}/grid.grpc.pb.h
 COMMAND ${RG_PROTOC_CMD} --proto_path=${PROTO_ROOT} --cpp_out=${GENERATED} --grpc_out=${GENERATED} --plugin=protoc-gen-grpc=${RG_GRPC_PLUGIN} ${PROTO_ROOT}/grid.proto
 DEPENDS ${PROTO_ROOT}/grid.proto VERBATIM)
add_library(grid_proto ${GENERATED}/grid.pb.cc ${GENERATED}/grid.grpc.pb.cc)
target_include_directories(grid_proto PUBLIC ${GENERATED})
target_link_libraries(grid_proto PUBLIC protobuf::libprotobuf gRPC::grpc++)
add_library(worker_core src/process.cpp src/executor.cpp src/log_pipe.cpp)
target_include_directories(worker_core PUBLIC include)
target_link_libraries(worker_core PUBLIC grid_proto OpenSSL::Crypto nlohmann_json::nlohmann_json Threads::Threads)
target_compile_definitions(worker_core PRIVATE _GNU_SOURCE)
target_compile_options(worker_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(regressgrid-worker src/main.cpp)
target_link_libraries(regressgrid-worker PRIVATE worker_core)
target_compile_options(regressgrid-worker PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(RG_SANITIZERS)
 foreach(target worker_core regressgrid-worker)
  target_compile_options(${target} PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(${target} PUBLIC -fsanitize=address,undefined)
 endforeach()
endif()
if(RG_TSAN)
 if(RG_SANITIZERS)
  message(FATAL_ERROR "RG_TSAN and RG_SANITIZERS are mutually exclusive")
 endif()
 foreach(target worker_core regressgrid-worker)
  target_compile_options(${target} PUBLIC -fsanitize=thread -fno-omit-frame-pointer -g)
  target_link_options(${target} PUBLIC -fsanitize=thread)
 endforeach()
endif()
if(RG_TIDY)
 find_program(CLANG_TIDY_EXE clang-tidy REQUIRED)
 set_target_properties(worker_core regressgrid-worker PROPERTIES CXX_CLANG_TIDY "${CLANG_TIDY_EXE};--warnings-as-errors=*")
endif()
add_library(rg_bench INTERFACE)
target_include_directories(rg_bench INTERFACE include)
target_compile_features(rg_bench INTERFACE cxx_std_20)
if(RG_BENCH)
 add_executable(kernel-bench bench/kernel_bench.cpp)
 target_link_libraries(kernel-bench PRIVATE rg_bench)
 target_compile_options(kernel-bench PRIVATE -O2 -Wall -Wextra -Wpedantic -Werror)
 if(RG_TSAN)
  target_compile_options(kernel-bench PRIVATE -fsanitize=thread)
  target_link_options(kernel-bench PRIVATE -fsanitize=thread)
 endif()
 add_executable(spsc-bench bench/spsc_bench.cpp)
 target_include_directories(spsc-bench PRIVATE include)
 target_link_libraries(spsc-bench PRIVATE Threads::Threads)
 target_compile_options(spsc-bench PRIVATE -O2 -Wall -Wextra -Wpedantic -Werror)
 if(RG_TSAN)
  target_compile_options(spsc-bench PRIVATE -fsanitize=thread)
  target_link_options(spsc-bench PRIVATE -fsanitize=thread)
 endif()
endif()
include(CTest)
if(BUILD_TESTING)
 find_package(GTest CONFIG REQUIRED)
 add_executable(worker-tests tests/process_test.cpp tests/spsc_ring_test.cpp tests/executor_log_test.cpp)
 target_link_libraries(worker-tests PRIVATE worker_core GTest::gtest_main)
 include(GoogleTest)
 gtest_discover_tests(worker-tests DISCOVERY_TIMEOUT 30)
 # Plain executable (no GoogleTest, no gRPC) for the header-only bench toolkit.
 add_executable(bench-selftest tests/bench_selftest.cpp)
 target_link_libraries(bench-selftest PRIVATE rg_bench)
 target_compile_options(bench-selftest PRIVATE -Wall -Wextra -Wpedantic -Werror)
 if(RG_SANITIZERS)
  target_compile_options(bench-selftest PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(bench-selftest PRIVATE -fsanitize=address,undefined)
 endif()
 if(RG_TSAN)
  target_compile_options(bench-selftest PRIVATE -fsanitize=thread -fno-omit-frame-pointer -g)
  target_link_options(bench-selftest PRIVATE -fsanitize=thread)
 endif()
 add_test(NAME bench.selftest COMMAND bench-selftest)
endif()
install(TARGETS regressgrid-worker RUNTIME DESTINATION bin)
