cmake_minimum_required(VERSION 3.22)
project(regressgrid_worker VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(RG_SANITIZERS "Enable address and undefined-behavior sanitizers" OFF)
option(RG_TIDY "Enable clang-tidy" OFF)
find_package(Threads REQUIRED)
find_package(gRPC CONFIG REQUIRED)
find_package(Protobuf REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
set(PROTO_ROOT ${CMAKE_CURRENT_SOURCE_DIR}/../proto)
set(GENERATED ${CMAKE_CURRENT_BINARY_DIR}/generated)
file(MAKE_DIRECTORY ${GENERATED})
add_custom_command(OUTPUT ${GENERATED}/grid.pb.cc ${GENERATED}/grid.pb.h ${GENERATED}/grid.grpc.pb.cc ${GENERATED}/grid.grpc.pb.h
 COMMAND protobuf::protoc --proto_path=${PROTO_ROOT} --cpp_out=${GENERATED} --grpc_out=${GENERATED} --plugin=protoc-gen-grpc=$<TARGET_FILE:gRPC::grpc_cpp_plugin> ${PROTO_ROOT}/grid.proto
 DEPENDS ${PROTO_ROOT}/grid.proto VERBATIM)
add_library(grid_proto ${GENERATED}/grid.pb.cc ${GENERATED}/grid.grpc.pb.cc)
target_include_directories(grid_proto PUBLIC ${GENERATED})
target_link_libraries(grid_proto PUBLIC protobuf::libprotobuf gRPC::grpc++)
add_library(worker_core src/process.cpp src/executor.cpp)
target_include_directories(worker_core PUBLIC include)
target_link_libraries(worker_core PUBLIC grid_proto OpenSSL::Crypto nlohmann_json::nlohmann_json Threads::Threads)
target_compile_definitions(worker_core PRIVATE _GNU_SOURCE)
target_compile_options(worker_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(regressgrid-worker src/main.cpp)
target_link_libraries(regressgrid-worker PRIVATE worker_core)
target_compile_options(regressgrid-worker PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(RG_SANITIZERS)
 foreach(target worker_core regressgrid-worker)
  target_compile_options(${target} PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(${target} PUBLIC -fsanitize=address,undefined)
 endforeach()
endif()
if(RG_TIDY)
 find_program(CLANG_TIDY_EXE clang-tidy REQUIRED)
 set_target_properties(worker_core regressgrid-worker PROPERTIES CXX_CLANG_TIDY "${CLANG_TIDY_EXE};--warnings-as-errors=*")
endif()
include(CTest)
if(BUILD_TESTING)
 find_package(GTest CONFIG REQUIRED)
 add_executable(worker-tests tests/process_test.cpp)
 target_link_libraries(worker-tests PRIVATE worker_core GTest::gtest_main)
 include(GoogleTest)
 gtest_discover_tests(worker-tests DISCOVERY_TIMEOUT 30)
endif()
install(TARGETS regressgrid-worker RUNTIME DESTINATION bin)
