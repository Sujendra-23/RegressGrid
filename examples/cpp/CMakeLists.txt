cmake_minimum_required(VERSION 3.22)
project(vector_kernel LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
include(CTest)
add_executable(vector-kernel main.cpp)
add_test(NAME vector.add COMMAND vector-kernel add)
add_test(NAME vector.dot COMMAND vector-kernel dot)
add_test(NAME vector.benchmark COMMAND vector-kernel benchmark)
install(TARGETS vector-kernel RUNTIME DESTINATION bin)
