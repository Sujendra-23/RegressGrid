cmake_minimum_required(VERSION 3.22)
project(alu_regression LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
include(CTest)
find_package(verilator HINTS $ENV{VERILATOR_ROOT} /usr/share/verilator /opt/homebrew/share/verilator REQUIRED)
add_executable(alu-sim sim.cpp)
verilate(alu-sim SOURCES alu.sv TRACE)
add_test(NAME alu.add COMMAND alu-sim add)
add_test(NAME alu.subtract COMMAND alu-sim subtract)
add_test(NAME alu.bitwise COMMAND alu-sim bitwise)
install(TARGETS alu-sim RUNTIME DESTINATION bin)
